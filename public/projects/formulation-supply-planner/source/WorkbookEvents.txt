Option Explicit
Private Sub Workbook_SheetChange(ByVal Sh As Object, ByVal Target As Range)
    If Sh.Name = "Supply Plan" Then
        If Not Intersect(Target, Sh.Range("B6:B8")) Is Nothing Then InvalidateResults "Supply Plan"
    ElseIf Sh.Name = "Cost Optimisation" Then
        If Not Intersect(Target, Sh.Range("B6:B8,E15:E26")) Is Nothing Then InvalidateResults "Cost Optimisation"
    ElseIf Sh.Name = "Other Stock" Then
        If Not Intersect(Target, Sh.Range("B7:B16")) Is Nothing Then InvalidateResults "Supply Plan"
        If Not Intersect(Target, Sh.Range("C7:C16")) Is Nothing Then InvalidateResults "Cost Optimisation"
    ElseIf Sh.Name = "Formulation Rules" Then
        If Not Intersect(Target, Sh.Range("B5:C19,C22:D33")) Is Nothing Then InvalidateResults
    ElseIf Sh.Name = "Model" Then
        If Not Intersect(Target, Sh.Range("C5:D16,G5:N16,B31:B34")) Is Nothing Then InvalidateResults
    End If
End Sub
